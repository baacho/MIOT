import RPi.GPIO as GPIO
import time

SERVO_PIN = 18

GPIO.setmode(GPIO.BCM)
GPIO.setup(SERVO_PIN, GPIO.OUT)

servo = GPIO.PWM(SERVO_PIN, 50)
servo.start(0)

current_angle = 90
total_movement = 0

def move_servo(angle):
    global current_angle, total_movement

    if angle < 0 or angle > 180:
        print("Angle must be between 0 and 180")
        return

    movement = angle - current_angle
    duty = 2.5 + (angle / 18)

    servo.ChangeDutyCycle(duty)
    time.sleep(0.5)
    servo.ChangeDutyCycle(0)

    total_movement += abs(movement)
    current_angle = angle

    rotations = total_movement / 360

    print("Current angle:", current_angle)
    print("Total angular movement:", total_movement)
    print("Estimated rotations:", rotations)

try:
    while True:
        print("\n1. Clockwise")
        print("2. Anticlockwise")
        print("3. Set angle")
        print("4. Exit")

        choice = input("Enter choice: ")

        if choice == "1":
            move_servo(180)

        elif choice == "2":
            move_servo(0)

        elif choice == "3":
            angle = float(input("Enter angle (0-180): "))
            move_servo(angle)

        elif choice == "4":
            break

        else:
            print("Invalid choice")

except KeyboardInterrupt:
    print("Program stopped")

finally:
    servo.stop()
    GPIO.cleanup()
