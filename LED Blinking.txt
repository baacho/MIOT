import RPi.GPIO as GPIO
import time

GPIO.setmode(GPIO.BCM)

LED = 17

GPIO.setup(LED, GPIO.OUT)

try:
    while True:
        GPIO.output(LED, GPIO.HIGH)
        print("LED ON")
        time.sleep(1)

        GPIO.output(LED, GPIO.LOW)
        print("LED OFF")
        time.sleep(1)

except KeyboardInterrupt:
    print("Program stopped")

finally:
    GPIO.cleanup()
